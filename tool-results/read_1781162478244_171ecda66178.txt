    71→export default function Home() {
    72→  const { toast } = useToast()
    73→  const fileInputRef = useRef<HTMLInputElement>(null)
    74→  const cardRef = useRef<HTMLDivElement>(null)
    75→  const previewWrapperRef = useRef<HTMLDivElement>(null)
    76→  const [isDragOver, setIsDragOver] = useState(false)
    77→  const [selectedBadge, setSelectedBadge] = useState('none')
    78→  const [isDownloading, setIsDownloading] = useState(false)
    79→  const [cardScale, setCardScale] = useState(1)
    80→  const [usedModel, setUsedModel] = useState<string | null>(null)
    81→
    82→  const {
    83→    productInput,
    84→    setProductInput,
    85→    isGeneratingText,
    86→    generatedContent,
    87→    generatedImages,
    88→    setGeneratingText,
    89→    setGeneratedContent,
    90→    updateContentField,
    91→    resetResults,
    92→  } = useGeneratorStore()
    93→
    94→  // Get model and token from model-store
    95→  const { currentModel, apiToken } = useModelStore()
    96→
    97→  const isGenerating = isGeneratingText
    98→  const hasResults = generatedContent || generatedImages.length > 0 || productInput.uploadedImage
    99→  const isCardVisible = generatedContent && productInput.uploadedImage
   100→
   101→  // Масштабирование карточки — пересчитываем при любом изменении размера и при появлении карточки
   102→  useEffect(() => {
   103→    const updateScale = () => {
   104→      if (previewWrapperRef.current) {
   105→        const containerWidth = previewWrapperRef.current.offsetWidth
   106→        setCardScale(containerWidth / 900)
   107→      }
   108→    }
   109→
   110→    updateScale()
   111→
   112→    const observer = new ResizeObserver(updateScale)
   113→    if (previewWrapperRef.current) {
   114→      observer.observe(previewWrapperRef.current)
   115→    }
   116→
   117→    return () => observer.disconnect()
   118→  }, [isCardVisible])
   119→
   120→  const handleImageUpload = useCallback(
   121→    (file: File) => {
   122→      if (!file.type.startsWith('image/')) {
   123→        toast({ title: 'Ошибка', description: 'Загрузите изображение (jpg, png, webp)', variant: 'destructive' })
   124→        return
   125→      }
   126→      if (file.size > 10 * 1024 * 1024) {
   127→        toast({ title: 'Ошибка', description: 'Размер файла не должен превышать 10 МБ', variant: 'destructive' })
   128→        return
   129→      }
   130→      const reader = new FileReader()
   131→      reader.onload = (e) => {
   132→        const base64 = e.target?.result as string
   133→        setProductInput({ uploadedImage: base64, uploadedImageName: file.name })
   134→      }
   135→      reader.readAsDataURL(file)
   136→    },
   137→    [setProductInput, toast]
   138→  )
   139→
   140→  const handleDrop = useCallback((e: React.DragEvent) => {
   141→    e.preventDefault()
   142→    setIsDragOver(false)
   143→    const file = e.dataTransfer.files[0]
   144→    if (file) handleImageUpload(file)
   145→  }, [handleImageUpload])
   146→
   147→  const handleDragOver = useCallback((e: React.DragEvent) => { e.preventDefault(); setIsDragOver(true) }, [])
   148→  const handleDragLeave = useCallback((e: React.DragEvent) => { e.preventDefault(); setIsDragOver(false) }, [])

   150→    const file = e.target.files?.[0]
   151→    if (file) handleImageUpload(file)
   152→  }, [handleImageUpload])
   153→
   154→  const removeUploadedImage = useCallback(() => {
   155→    setProductInput({ uploadedImage: null, uploadedImageName: null })
   156→    if (fileInputRef.current) fileInputRef.current.value = ''
   157→  }, [setProductInput])
   158→
   159→  const productImageForCard = productInput.uploadedImage
   160→
   161→  const handleGenerate = useCallback(async () => {
   162→    if (!productInput.productName.trim()) {
   163→      toast({ title: 'Укажите название товара', description: 'Название обязательно для генерации', variant: 'destructive' })
   164→      return
   165→    }
   166→    if (!productInput.uploadedImage) {
   167→      toast({ title: 'Загрузите фото товара', description: 'Для генерации карточки необходимо загрузить фото', variant: 'destructive' })
   168→      return
   169→    }
   170→
   171→    setGeneratingText(true)
   172→    setUsedModel(null)
   173→    try {
   174→      const textRes = await fetch('/api/generate-text', {
   175→        method: 'POST',
   176→        headers: { 'Content-Type': 'application/json' },
   177→        body: JSON.stringify({
   178→          productName: productInput.productName,
   179→          category: productInput.category,
   180→          sellingPoints: productInput.sellingPoints,
   181→          targetAudience: productInput.targetAudience,
   182→          keywords: productInput.keywords,
   183→          style: productInput.style,
   184→          model: currentModel,
   185→          apiToken: apiToken || undefined,
   186→        }),
   187→      })
   188→
   189→      let textData
   190→      try { textData = await textRes.json() } catch { throw new Error('Сервер вернул некорректный ответ') }
   191→      if (!textRes.ok) throw new Error(textData?.error || 'Ошибка генерации текста')
   192→
   193→      // Extract used model from response header
   194→      const modelUsed = textRes.headers?.get('X-Model-Used')
   195→      if (modelUsed) {
   196→        setUsedModel(modelUsed)
   197→      }
   198→
   199→      setGeneratedContent(textData)
   200→      toast({
   201→        title: 'Карточка готова',
   202→        description: modelUsed && modelUsed !== currentModel
   203→          ? `Использована модель: ${modelUsed.split('/').pop()?.replace(':free', '')}`
   204→          : 'Текст сгенерирован, карточка готова к скачиванию',
   205→      })
   206→    } catch (err) {
   207→      toast({ title: 'Ошибка генерации', description: err instanceof Error ? err.message : 'Попробуйте ещё раз', variant: 'destructive' })
   208→    } finally {
   209→      setGeneratingText(false)
   210→    }
   211→  }, [productInput, setGeneratingText, setGeneratedContent, toast, currentModel, apiToken])
   212→
   213→  const handleDownloadCard = useCallback(async () => {
   214→    if (!cardRef.current) return
   215→    setIsDownloading(true)
   216→    try {
   217→      const domToImage = (await import('dom-to-image-more')).default
   218→      const dataUrl = await domToImage.toPng(cardRef.current, {
   219→        width: 900, height: 1200, quality: 1,
   220→        style: { transform: 'none' },
   221→      })
   222→      const link = document.createElement('a')
   223→      link.href = dataUrl
   224→      link.download = `${productInput.productName || 'product'}-card.png`
   225→      document.body.appendChild(link)
   226→      link.click()
   227→      document.body.removeChild(link)
   228→      toast({ title: 'Скачано', description: 'Карточка товара сохранена' })
   229→    } catch {
   230→      toast({ title: 'Ошибка скачивания', description: 'Не удалось сохранить карточку', variant: 'destructive' })
   231→    } finally {
   232→      setIsDownloading(false)
   233→    }
   234→  }, [productInput.productName, toast])
   235→
   236→  const handleRegenerate = useCallback(() => {
   237→    useGeneratorStore.setState({ generatedImages: [], selectedImageIndex: 0, completedImageCount: 0 })
   238→    handleGenerate()
   239→  }, [handleGenerate])
   240→
   241→  return (
   242→    <div className="min-h-screen bg-background flex flex-col">
   243→      {/* Header */}
   244→      <header className="sticky top-0 z-40 border-b bg-card/80 backdrop-blur-sm">
   245→        <div className="mx-auto flex h-12 sm:h-14 md:h-16 max-w-7xl items-center justify-between px-3 sm:px-5 md:px-6 lg:px-8">
   246→          <div className="flex items-center gap-2 sm:gap-3">
   247→            <div className="flex h-8 w-8 sm:h-9 sm:w-9 items-center justify-center rounded-lg bg-primary text-primary-foreground">
   248→              <ShoppingBag className="h-4 w-4 sm:h-5 sm:w-5" />
   249→            </div>
   250→            <div>
   251→              <h1 className="text-sm sm:text-base md:text-lg font-bold leading-tight">E-Commerce Генератор</h1>
   252→              <p className="hidden md:block text-xs text-muted-foreground">Готовые карточки для маркетплейсов</p>
   253→            </div>
   254→          </div>
   255→          <div className="flex items-center gap-1.5 sm:gap-2">
   256→            <ModelSelector />
   257→            <ThemeToggle />
   258→            {hasResults && (
   259→              <Button variant="outline" size="sm" onClick={resetResults} className="gap-1.5 text-xs h-8 sm:h-9">
   260→                <RefreshCw className="h-3 w-3" />
   261→                <span className="hidden sm:inline">Начать заново</span>
   262→                <span className="sm:hidden">Заново</span>
   263→              </Button>
   264→            )}
   265→          </div>
   266→        </div>
   267→      </header>
   268→
   269→      {/* Main Content */}
   270→      <main className="flex-1 mx-auto w-full max-w-7xl px-3 py-3 sm:px-5 sm:py-4 md:px-6 md:py-6 lg:px-8 lg:py-8">
   271→        <div className="grid gap-3 sm:gap-4 md:gap-6 lg:grid-cols-2">
   272→          {/* LEFT PANEL — Input Form */}
   273→          <motion.div
   274→            initial={{ opacity: 0, y: 10 }}
   275→            animate={{ opacity: 1, y: 0 }}
   276→            transition={{ duration: 0.3 }}
   277→          >
   278→            <Card>
   279→              <CardHeader className="pb-3 sm:pb-4 px-4 sm:px-6 pt-4 sm:pt-6">
   280→                <CardTitle className="flex items-center gap-2 text-base sm:text-lg">
   281→                  <Type className="h-4 w-4 sm:h-5 sm:w-5 text-primary" />
   282→                  Данные о товаре
   283→                </CardTitle>
   284→                <CardDescription className="text-xs sm:text-sm">
   285→                  Заполните информацию — на основе неё сгенерируется готовая карточка
   286→                </CardDescription>
   287→              </CardHeader>
   288→              <CardContent className="space-y-3 sm:space-y-4 md:space-y-5 px-4 sm:px-6 pb-4 sm:pb-6">
   289→                {/* Product Image Upload */}
   290→                <div className="space-y-1.5 sm:space-y-2">
   291→                  <Label className="text-xs sm:text-sm">Фото товара <span className="text-destructive font-normal">* обязательно</span></Label>
   292→                  {!productInput.uploadedImage ? (
   293→                    <div
   294→                      onDrop={handleDrop}
   295→                      onDragOver={handleDragOver}
   296→                      onDragLeave={handleDragLeave}
   297→                      onClick={() => fileInputRef.current?.click()}
   298→                      className={`flex cursor-pointer flex-col items-center justify-center rounded-lg border-2 border-dashed p-4 sm:p-6 transition-colors ${
   299→                        isDragOver
   300→                          ? 'border-primary bg-primary/5'
   301→                          : 'border-muted-foreground/25 hover:border-primary/50 hover:bg-muted/50'
   302→                      }`}
   303→                    >
   304→                      <Upload className={`mb-2 h-6 w-6 sm:h-8 sm:w-8 ${isDragOver ? 'text-primary' : 'text-muted-foreground'}`} />
   305→                      <p className="text-xs sm:text-sm font-medium text-muted-foreground">
   306→                        Перетащите фото или нажмите
   307→                      </p>
   308→                      <p className="mt-1 text-[10px] sm:text-xs text-muted-foreground/70">
   309→                        JPG, PNG, WebP (до 10 МБ)
   310→                      </p>
   311→                    </div>
   312→                  ) : (
   313→                    <div className="relative inline-block">
   314→                      <img
   315→                        src={productInput.uploadedImage}
   316→                        alt="Фото товара"
   317→                        className="h-20 sm:h-28 md:h-32 w-auto max-w-full rounded-lg border object-contain"
   318→                      />
   319→                      <Button variant="destructive" size="icon" className="absolute -right-2 -top-2 h-6 w-6" onClick={removeUploadedImage}>
   320→                        <X className="h-3 w-3" />
   321→                      </Button>
   322→                      <p className="mt-1 text-[10px] sm:text-xs text-muted-foreground truncate max-w-[200px] sm:max-w-none">
   323→                        {productInput.uploadedImageName}
   324→                      </p>
   325→                    </div>
   326→                  )}
   327→                  <input ref={fileInputRef} type="file" accept="image/jpeg,image/png,image/webp" className="hidden" onChange={handleFileInput} />
   328→                </div>
   329→
   330→                {/* Product Name */}
   331→                <div className="space-y-1.5 sm:space-y-2">
   332→                  <Label htmlFor="productName" className="text-xs sm:text-sm">
   333→                    Название товара <span className="text-destructive">*</span>
   334→                  </Label>
   335→                  <Input
   336→                    id="productName"
   337→                    placeholder='Беспроводные наушники SoundMax Pro'
   338→                    value={productInput.productName}
   339→                    onChange={(e) => setProductInput({ productName: e.target.value })}
   340→                    className="text-sm"
   341→                  />
   342→                </div>
   343→
   344→                {/* Category + Style */}
   345→                <div className="grid grid-cols-1 sm:grid-cols-2 gap-2 sm:gap-3">
   346→                  <div className="space-y-1.5 sm:space-y-2">
   347→                    <Label className="text-xs sm:text-sm">Категория</Label>
   348→                    <Select value={productInput.category} onValueChange={(value) => setProductInput({ category: value })}>
   349→                      <SelectTrigger className="w-full text-sm"><SelectValue placeholder="Выберите" /></SelectTrigger>
   350→                      <SelectContent>
   351→                        {CATEGORIES.map((cat) => (<SelectItem key={cat.value} value={cat.value}>{cat.label}</SelectItem>))}
   352→                      </SelectContent>
   353→                    </Select>
   354→                  </div>
   355→                  <div className="space-y-1.5 sm:space-y-2">
   356→                    <Label className="text-xs sm:text-sm">Стиль карточки</Label>
   357→                    <Select value={productInput.style} onValueChange={(value) => setProductInput({ style: value })}>
   358→                      <SelectTrigger className="w-full text-sm"><SelectValue placeholder="Выберите" /></SelectTrigger>
   359→                      <SelectContent>
   360→                        {STYLES.map((s) => (<SelectItem key={s.value} value={s.value}>{s.label}</SelectItem>))}
   361→                      </SelectContent>
   362→                    </Select>
   363→                  </div>
   364→                </div>
   365→
   366→                {/* Key Selling Points */}
   367→                <div className="space-y-1.5 sm:space-y-2">
   368→                  <Label htmlFor="sellingPoints" className="text-xs sm:text-sm">Ключевые преимущества</Label>
   369→                  <Textarea
   370→                    id="sellingPoints"
   371→                    placeholder={"Активное шумоподавление\nВремя работы 30 часов\nВодозащита IPX5"}
   372→                    value={productInput.sellingPoints}
   373→                    onChange={(e) => setProductInput({ sellingPoints: e.target.value })}
   374→                    rows={3}
   375→                    className="text-sm"
   376→                  />
   377→                  <p className="text-[10px] sm:text-xs text-muted-foreground">Каждое преимущество с новой строки</p>
   378→                </div>
   379→
   380→                {/* Target Audience + Keywords */}
   381→                <div className="grid grid-cols-1 sm:grid-cols-2 gap-2 sm:gap-3">
   382→                  <div className="space-y-1.5 sm:space-y-2">
   383→                    <Label htmlFor="targetAudience" className="text-xs sm:text-sm">Целевая аудитория</Label>
   384→                    <Input id="targetAudience" placeholder='женщины 25-35' value={productInput.targetAudience} onChange={(e) => setProductInput({ targetAudience: e.target.value })} className="text-sm" />
   385→                  </div>
   386→                  <div className="space-y-1.5 sm:space-y-2">
   387→                    <Label htmlFor="keywords" className="text-xs sm:text-sm">Ключевые слова</Label>
   388→                    <Input id="keywords" placeholder='наушники, звук' value={productInput.keywords} onChange={(e) => setProductInput({ keywords: e.target.value })} className="text-sm" />
   389→                  </div>
   390→                </div>
   391→
   392→                {/* Badge */}
   393→                <div className="space-y-1.5 sm:space-y-2">
   394→                  <Label className="text-xs sm:text-sm">Бейдж на карточке</Label>
   395→                  <Select value={selectedBadge} onValueChange={setSelectedBadge}>
   396→                    <SelectTrigger className="w-full text-sm"><SelectValue /></SelectTrigger>
   397→                    <SelectContent>
   398→                      {BADGE_OPTIONS.map((b) => (<SelectItem key={b.value} value={b.value}>{b.label}</SelectItem>))}
   399→                    </SelectContent>
   400→                  </Select>
   401→                </div>
   402→
   403→                <Separator />
   404→
   405→                {/* Generate Button */}
   406→                <Button
   407→                  size="lg"
   408→                  className="w-full gap-2 text-sm sm:text-base h-11 sm:h-12"
   409→                  onClick={handleGenerate}
   410→                  disabled={isGenerating || !productInput.productName.trim()}
   411→                >
   412→                  {isGenerating ? (
   413→                    <><Loader2 className="h-4 w-4 sm:h-5 sm:w-5 animate-spin" />Генерация...</>
   414→                  ) : (
   415→                    <><Sparkles className="h-4 w-4 sm:h-5 sm:w-5" />Сгенерировать карточку</>
   416→                  )}
   417→                </Button>
   418→
   419→                {/* Model info hint */}
   420→                {usedModel && usedModel !== currentModel && (
   421→                  <p className="text-[10px] text-muted-foreground text-center">
   422→                    Использована резервная модель: {usedModel.split('/').pop()?.replace(':free', '')}
   423→                  </p>
   424→                )}
   425→              </CardContent>
   426→            </Card>
   427→
   428→            {/* Editable Text Section */}
   429→            {generatedContent && (
   430→              <Card className="mt-3 sm:mt-4 md:mt-6">
   431→                <CardHeader className="pb-2 sm:pb-3 px-4 sm:px-6 pt-4 sm:pt-6">
   432→                  <CardTitle className="flex items-center gap-2 text-sm sm:text-base">
   433→                    <Type className="h-3.5 w-3.5 sm:h-4 sm:w-4 text-primary" />
   434→                    Редактировать текст карточки
   435→                  </CardTitle>
   436→                  <CardDescription className="text-xs">Изменения сразу отображаются на карточке</CardDescription>
   437→                </CardHeader>
   438→                <CardContent className="space-y-2 sm:space-y-3 px-4 sm:px-6 pb-4 sm:pb-6">
   439→                  <div className="space-y-1">
   440→                    <Label className="text-[10px] sm:text-xs uppercase tracking-wider text-muted-foreground">Заголовок</Label>
   441→                    <Input value={generatedContent.title} onChange={(e) => updateContentField('title', e.target.value)} className="font-semibold text-sm" />

   443→                  <div className="space-y-1">
   444→                    <Label className="text-[10px] sm:text-xs uppercase tracking-wider text-muted-foreground">Подзаголовок</Label>
   445→                    <Input value={generatedContent.subtitle} onChange={(e) => updateContentField('subtitle', e.target.value)} className="text-sm" />
   446→                  </div>
   447→                  <div className="space-y-1">
   448→                    <Label className="text-[10px] sm:text-xs uppercase tracking-wider text-muted-foreground">Преимущества</Label>
   449→                    <div className="flex flex-wrap gap-1.5 sm:gap-2">
   450→                      {generatedContent.features.map((feature, index) => (
   451→                        <Badge key={index} variant="secondary" className="gap-1 px-2 py-1 sm:px-3 sm:py-1.5 text-xs sm:text-sm">
   452→                          <CheckCircle2 className="h-2.5 w-2.5 sm:h-3 sm:w-3 text-primary" />
   453→                          <input
   454→                            value={feature}
